  <div class="card">
    <h2>Cos'è un audit di processo</h2>
    <p>Verifica sistematica per stabilire se le attività produttive sono <b>conformi ai requisiti pianificati</b> e se tali requisiti sono <b>attuati efficacemente</b>.</p>
    <div class="refbox">📘 ISO 19011:2018 + ISO 9001:2015 cl. 8.5 / IATF 16949 cl. 8.5</div>
  </div>

  <div class="card">
    <h2>🗺️ Percorso consigliato sul campo</h2>
    <p style="margin-bottom:12px;color:#555">Segui il flusso del prodotto <b>da monte a valle</b>: ogni area prepara il terreno per quella successiva. Riduce i backtrack e permette di verificare la tracciabilità in modo naturale.</p>
    <ol class="step-list">
      <li><div class="step-num">1</div><div class="step-body">
        <div class="step-title">📦 Magazzino materie prime / Incoming</div>
        <div class="step-desc">Identificazione lotti, FIFO, etichettature, stato accettazione. Materiali NC in quarantena. Tracciabilità: contenitore → DDT → fornitore.</div>
        <span class="step-tip">💡 "Da quanto è qui questo lotto?" / "Come sapete che è stato accettato?"</span>
      </div></li>
      <li><div class="step-num">2</div><div class="step-body">
        <div class="step-title">📋 Ufficio qualità / Documentazione</div>
        <div class="step-desc">Control Plan, FMEA, istruzioni di lavoro: revisione aggiornata? Coerenti? Prendi nota delle versioni per confrontarle con quelle in linea.</div>
        <span class="step-tip">💡 Non farlo in linea: serve la versione "ufficiale" prima di osservare la realtà.</span>
      </div></li>
      <li><div class="step-num">3</div><div class="step-body">
        <div class="step-title">⚙️ Prima stazione di processo (avvio linea)</div>
        <div class="step-desc">Setup, primo pezzo, first-off approvato, parametri impostati. Operatore formato? Istruzioni nella revisione corretta?</div>
        <span class="step-tip">💡 Se possibile assisti a un cambio produzione: momento più critico.</span>
      </div></li>
      <li><div class="step-num">4</div><div class="step-body">
        <div class="step-title">🏭 Linea produttiva (a campione)</div>
        <div class="step-desc">1–2 cicli completi su stazioni diverse. Confronta visto vs istruzione. Poka-yoke, parametri critici, registrazioni. Intervista un operatore (non il capo).</div>
        <span class="step-tip">💡 Stai zitto e osserva: 3–5 minuti in silenzio valgono più di 10 domande teoriche.</span>
      </div></li>
      <li><div class="step-num">5</div><div class="step-body">
        <div class="step-title">🔬 Postazione controllo qualità</div>
        <div class="step-desc">Controlli del Control Plan rispettati? Strumenti tarati? Registrazioni complete, leggibili, firmate? Carte SPC aggiornate?</div>
        <span class="step-tip">💡 Prendi fogli registrazione ultime 4–8 ore: campioni mancanti? Valori al limite gestiti?</span>
      </div></li>
      <li><div class="step-num">6</div><div class="step-body">
        <div class="step-title">🔧 Area manutenzione / Magazzino attrezzature</div>
        <div class="step-desc">Piano manutenzione preventiva rispettato? Strumenti con taratura in scadenza? Area NC per utensili fuori uso identificata?</div>
        <span class="step-tip">💡 Cerca strumenti senza etichetta di taratura o con etichetta scaduta in uso: NC immediata.</span>
      </div></li>
      <li><div class="step-num">7</div><div class="step-body">
        <div class="step-title">🚫 Area non conformi</div>
        <div class="step-desc">Materiali in quarantena identificati? Tempo permanenza ragionevole? Decisione documentata? NC aperte da settimane senza decisione = NC.</div>
        <span class="step-tip">💡 Una NC senza decisione di disposizione dopo 7+ giorni è spesso una NC documentale.</span>
      </div></li>
      <li><div class="step-num">8</div><div class="step-body">
        <div class="step-title">📤 Magazzino prodotto finito / Spedizione</div>
        <div class="step-desc">Tracciabilità avanti completa: dal lotto materia prima al codice spedito. Prodotti identificati con stato qualità. Condizioni di conservazione idonee?</div>
        <span class="step-tip">💡 Chiudi il loop: prendi il pezzo dal punto 1 e arriva fino a qui con la catena lotti.</span>
      </div></li>
      <li><div class="step-num">9</div><div class="step-body">
        <div class="step-title">📊 Visual management / KPI board</div>
        <div class="step-desc">KPI aggiornati? Operatori li conoscono? Azioni correttive da audit/reclami precedenti implementate con evidenza di efficacia?</div>
        <span class="step-tip">💡 "Qual è il target FTQ di questa settimana?" — la risposta dice tutto.</span>
      </div></li>
    </ol>
    <div class="refbox">⏱ Tempo tipico: 3–4h. Pianifica 30 min opening, 2–2.5h campo, 30 min note, 30 min closing.</div>
  </div>

  <div class="card">
    <h2>Fasi dell'audit</h2>
    <h3>1. Pianificazione (–1 settimana)</h3>
    <ul><li>Scopo, obiettivi, criteri (norma + procedure + specifiche cliente)</li><li>Raccogli: flow chart, Control Plan, FMEA, istruzioni, NC aperte, KPI 3 mesi</li><li>Comunica data, durata e perimetro al responsabile</li></ul>
    <h3>2. Opening meeting (15 min)</h3>
    <ul><li>Presenta scopo e perimetro — non è ispezione personale</li><li>Conferma percorso e referenti per ogni area</li></ul>
    <h3>3. Esecuzione (vedi percorso sopra)</h3>
    <ul><li><b>Turtle</b>: Input / Output / Con chi / Con cosa / Come / KPI</li><li><b>4 evidenze</b>: osserva + intervista + documenti + misura campione</li><li>Annota solo <b>evidenze oggettive</b> (lotti, codici, ore, numeri)</li></ul>
    <h3>4. Closing meeting (30 min)</h3>

const STORAGE_PREFIX = 'audit_v5:';
const STORAGE_AUDIT_KEY = STORAGE_PREFIX + 'audit:';
let storageOK = false;

function storageTest() {
  try {
    const k = STORAGE_PREFIX + '__test__';
    localStorage.setItem(k, '1');
    const v = localStorage.getItem(k);
    localStorage.removeItem(k);
    return v === '1';
  } catch (e) {
    return false;
  }
}

function lsSet(key, obj) {
  try {
    localStorage.setItem(STORAGE_AUDIT_KEY + key, JSON.stringify(obj));
    return true;
  } catch (e) {
    console.error('lsSet failed:', e);
    return false;
  }
}

function lsGet(key) {
  try {
    const v = localStorage.getItem(STORAGE_AUDIT_KEY + key);
    return v ? JSON.parse(v) : null;
  } catch (e) {
    return null;
  }
}

function lsDel(key) {
  try { localStorage.removeItem(STORAGE_AUDIT_KEY + key); } catch (e) {}
}

function lsKeys() {
  const keys = [];
  try {
    for (let i = 0; i < localStorage.length; i++) {
      const k = localStorage.key(i);
      if (k && k.startsWith(STORAGE_AUDIT_KEY)) {
        keys.push(k.slice(STORAGE_AUDIT_KEY.length));
      }
    }
  } catch (e) {}
  return keys;
}

function getAllAudits() {
  return lsKeys()
    .map(k => lsGet(k))
    .filter(Boolean)
    .sort((a, b) => (b.savedAt || '').localeCompare(a.savedAt || ''));
}

function countAudits() { return lsKeys().length; }

function getStorageUsage() {
  let bytes = 0;
  try {
    for (let i = 0; i < localStorage.length; i++) {
      const k = localStorage.key(i);
      if (k && k.startsWith(STORAGE_PREFIX)) {
        bytes += (localStorage.getItem(k) || '').length + k.length;
      }
    }
  } catch (e) {}
  return bytes;
}

function readFormData() {
const phases = getPhases(currentNorm);
const prefix = currentNorm === ‘IATF16949’ ? ‘I’ : ‘S’;
const data = {
id: currentId || genId(),
savedAt: new Date().toISOString(),

answers: {}, notes: {},
conclusion: gv(‘r-conclusion’), followup: gv(‘r-followup’)
};
phases.forEach((p, pi) => p.items.forEach((_, qi) => {

const nt = document.querySelector(‘textarea[data-norm=”’ + currentNorm + ‘”][data-q=”’ + pi + ‘-’ + qi + ‘”]’);
if (nt && nt.value.trim()) data.notes[pi + ‘-’ + qi] = nt.value;
}));
return data;
}

/* ═══════════════════════════════════════════════════════════════════
CALCOLA STATS
═══════════════════════════════════════════════════════════════════ */
function calcStats(data) {
const phases = getPhases(data.norm || ‘ISO9001’);
let c = 0, m = 0, g = 0, na = 0, tot = 0;
phases.forEach((ph, pi) => ph.items.forEach((_, qi) => {
tot++;
const v = (data.answers || {})[pi + ‘-’ + qi];
if (v === ‘C’) c++;
else if (v === ‘NC-M’) m++;
else if (v === ‘NC-G’) g++;
else if (v === ‘NA’) na++;
}));
const done = c + m + g + na;
return { c, m, g, na, tot, done, missing: tot - done, pct: tot ? Math.round(done / tot * 100) : 0 };
}

}

function dl(content, name, type) {
try {
const blob = new Blob([’\uFEFF’ + content], { type });
const url = URL.createObjectURL(blob);
const a = document.createElement(‘a’);
a.href = url;
a.download = name;
a.style.display = ‘none’;
document.body.appendChild(a);
a.click();
setTimeout(() => {
document.body.removeChild(a);
URL.revokeObjectURL(url);
}, 1000);
return true;
} catch (e) {
console.error(‘Download failed:’, e);
showToast(’❌ Errore download: ’ + e.message, 4000);
return false;
}
}
